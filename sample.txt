import ollama

# Generate Python function using Ollama
def generate_function():
    prompt = """
    Write a Python function named sum_even_numbers
    that takes an array of numbers as an argument
    and returns the sum of all even numbers.

    Return ONLY the Python function.
    Do not include explanations or markdown.
    """

    response = ollama.generate(
        model="qwen2.5-coder:7b",
        prompt=prompt
    )

    code = response["response"].strip()

    # Remove Markdown code block markers
    if code.startswith("```"):
        code = code.split("\n", 1)[1]
        code = code.rsplit("```", 1)[0]

    # Return the generated code
    return code.strip()


# Test the generated function
def test_function(code):
    namespace = {}

    try:
        # Execute the generated Python code
        exec(code, namespace)

        # Get the generated function
        sum_even_numbers = namespace["sum_even_numbers"]

        # Sample test cases
        test_cases = [
            ([1, 2, 3, 4, 5, 6], 12),
            ([2, 4, 6, 8], 20),
            ([1, 3, 5, 7], 0),
            ([-2, -4, 3, 5], -6),
            ([], 0)
        ]

        print("\nTesting Generated Function")
        print("-" * 40)

        passed = 0

        for arr, expected in test_cases:
            result = sum_even_numbers(arr)

            print(
                f"Input: {arr} | "
                f"Actual: {result} | "
                f"Expected: {expected}"
            )

            if result == expected:
                passed += 1
                print("PASS")
            else:
                print("FAIL")

        print("-" * 40)
        print(f"Test Cases Passed: {passed}/{len(test_cases)}")

        if passed == len(test_cases):
            print("Overall Result: CORRECT")
        else:
            print("Overall Result: INCORRECT")

    except Exception as e:
        print("Error:", e)


# Main program
print("Ollama Python Function Generator")
print("=" * 40)

generated_code = generate_function()

print("\nGenerated Python Function:")
print("-" * 40)
print(generated_code)
print("-" * 40)

test_function(generated_code)

# 2. GENERATE CODE USING CODEGEMMA

def generate_code():

    prompt = """
    Generate ONLY a Python function named insertion_sort(arr).

    Requirements:
    1. Implement Insertion Sort manually.
    2. Sort the list in ascending order.
    3. Return the sorted list.
    4. Use loops and element comparison.
    5. Use the while loop to shift elements.
    6. Do NOT use sort().
    7. Do NOT use sorted().
    8. Do not provide explanations.
    9. Do not use Markdown.
    """

    response = ollama.chat(
        model="codegemma:2b",
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    )

    raw_code = response["message"]["content"]

    print("\nRAW CODEGEMMA RESPONSE")
    print("-" * 50)
    print(raw_code)

    # 3. CLEAN THE GENERATED CODE

    code = raw_code.replace("```python", "")
    code = code.replace("```", "")

    # Find the insertion_sort function
    start = code.find("def insertion_sort")

    if start == -1:
        return ""

    code = code[start:]

    return code.strip()


# 4. SYNTAX CHECKING

def check_syntax(code):

    try:
        ast.parse(code)
        return True

    except SyntaxError as e:
        print("\nSyntax Error:", e)
        return False


# 5. FUNCTIONAL TESTING

def test_insertion_sort(code):

    namespace = {}

    try:

        exec(code, namespace)

        if "insertion_sort" not in namespace:

        insertion_sort = namespace["insertion_sort"]

        # Sample test cases
        test_cases = [
            ([5, 3, 8, 4, 2], [2, 3, 4, 5, 8]),
            ([10, 1, 7, 3], [1, 3, 7, 10]),
            ([5, 5, 2, 1], [1, 2, 5, 5]),

            ([1], [1]),
            ([], []),
            ([1, 2, 3, 4], [1, 2, 3, 4]),
            ([-5, 3, -1, 2], [-5, -1, 2, 3])
        ]

        print("-" * 50)

# 6. ALGORITHM CHECKING

def check_algorithm(code):

    code_lower = code.lower()

    # Built-in sorting is not allowed
    if ".sort(" in code_lower:
        return False

    if "sorted(" in code_lower:
        return False

    # Check Insertion Sort characteristics
    has_loop = "for" in code_lower
    has_while = "while" in code_lower
    has_condition = "if" in code_lower
    has_comparison = ">" in code or "<" in code
    has_function = "def insertion_sort" in code_lower

    if (has_loop and has_while and
            has_condition and has_comparison and has_function):
        return True

    return False


# 7. SIMILARITY CALCULATION

def calculate_similarity(generated_code):

    generated_tokens = set(
        re.findall(r"[a-zA-Z_]+", generated_code.lower())
    )

    if len(generated_tokens) == 0:
        return 0

    best_similarity = 0

    for real_code in real_code_dataset:

        real_tokens = set(
            re.findall(r"[a-zA-Z_]+", real_code.lower())
        )

        common_tokens = generated_tokens.intersection(real_tokens)
        union_tokens = generated_tokens.union(real_tokens)

        if len(union_tokens) == 0:
            similarity = 0
        else:
            similarity = len(common_tokens) / len(union_tokens)

        if similarity > best_similarity:
            best_similarity = similarity

    return round(best_similarity * 100, 2)


# 8. DISCRIMINATOR

def discriminator(code):

    # Syntax checking
    syntax = check_syntax(code)

    if not syntax:
        return {
            "syntax": False,
            "functional": False,
            "algorithm": False,
            "similarity": 0,
            "score": 0
        }

    # Functional testing
    functional = test_insertion_sort(code)

    # Algorithm checking
    algorithm = check_algorithm(code)

    # Similarity checking
    similarity = calculate_similarity(code)

    # Calculate final score
    score = 0

    if syntax:
        score += 20

    if functional:
        score += 40

    if algorithm:
        score += 20

    # Similarity contributes 20 marks
    similarity_score = similarity * 0.20

    score += similarity_score

    return {
        "syntax": syntax,
        "functional": functional,
        "algorithm": algorithm,
        "similarity": similarity,
        "score": round(score, 2)
    }
