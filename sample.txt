import ollama

# Generate Python function using Ollama
def generate_function():
    prompt = """
    Write a Python function named sum_even_numbers
    that takes an array of numbers as an argument
    and returns the sum of all even numbers.

    Return ONLY the Python function.
    Do not include explanations or markdown.
    """

    response = ollama.generate(
        model="qwen2.5-coder:7b",
        prompt=prompt
    )

    code = response["response"].strip()

    # Remove Markdown code block markers
    if code.startswith("```"):
        code = code.split("\n", 1)[1]
        code = code.rsplit("```", 1)[0]

    # Return the generated code
    return code.strip()


# Test the generated function
def test_function(code):
    namespace = {}

    try:
        # Execute the generated Python code
        exec(code, namespace)

        # Get the generated function
        sum_even_numbers = namespace["sum_even_numbers"]

        # Sample test cases
        test_cases = [
            ([1, 2, 3, 4, 5, 6], 12),
            ([2, 4, 6, 8], 20),
            ([1, 3, 5, 7], 0),
            ([-2, -4, 3, 5], -6),
            ([], 0)
        ]

        print("\nTesting Generated Function")
        print("-" * 40)

        passed = 0

        for arr, expected in test_cases:
            result = sum_even_numbers(arr)

            print(
                f"Input: {arr} | "
                f"Actual: {result} | "
                f"Expected: {expected}"
            )

            if result == expected:
                passed += 1
                print("PASS")
            else:
                print("FAIL")

        print("-" * 40)
        print(f"Test Cases Passed: {passed}/{len(test_cases)}")

        if passed == len(test_cases):
            print("Overall Result: CORRECT")
        else:
            print("Overall Result: INCORRECT")

    except Exception as e:
        print("Error:", e)


# Main program
print("Ollama Python Function Generator")
print("=" * 40)

generated_code = generate_function()

print("\nGenerated Python Function:")
print("-" * 40)
print(generated_code)
print("-" * 40)

test_function(generated_code)

# --------------------------------
# MODEL CONFIGURATION
# --------------------------------

model = "qwen2.5-coder:7b"

prompts = """
Generate a Python program to implement selection sort
with function name selection_sort().

Requirements:
1. Implement selection sort manually.
2. Do not use sorted() or .sort().
3. Sort the array in ascending order.
4. Return the sorted array.
5. Do not include comments or explanations.
6. Generate only the function.
"""


# --------------------------------
# 1. CODE GENERATION
# --------------------------------

def generate():

    c = ollama.generate(
        model=model,
        prompt=prompts
    )["response"]

    c = c.strip().replace("```python", "").replace("```", "")

    print("\nGenerated Code:\n")
    print(c)

    with open("newcodes.txt", "w") as f:
        f.write(c)

    global codes
    codes = c

    print("\nCode saved to newcodes.txt")


# --------------------------------
# 2. TEST CASE TESTING
# --------------------------------

def testcase_testing():

    print("\nTest Case Testing")

    ns = {}
    exec(codes, ns)

    testcases = [
        ([5, 2, 4, 3, 1], [1, 2, 3, 4, 5]),
        ([3, 4, 2, 1, 2], [1, 2, 2, 3, 4]),
        ([78, 43, 67, 34, 21], [21, 34, 43, 67, 78])
    ]

    count = 0

    for i, (arr, actual) in enumerate(testcases, 1):

        expected = ns["selection_sort"](arr.copy())

        if expected == actual:
            count += 1
            print(f"Test case {i}: PASS")
        else:
            print(f"Test case {i}: FAIL")

    print("Passed:", count)
    print("Failed:", len(testcases) - count)


# --------------------------------
# 3. SYNTAX CHECKING
# --------------------------------

def checksyntax():

    print("\nSyntax Checking")

    try:
        ast.parse(codes)
        print("Syntax is correct")

    except SyntaxError:
        print("Syntax error found")


# --------------------------------
# 4. ALGORITHM CHECKING
# --------------------------------

def checkalgo():

    print("\nAlgorithm Checking")

    if (
        "for" in codes
        and "if" in codes
        and ".sort(" not in codes
        and "sorted(" not in codes
    ):
        print("Algorithm check passed")

    else:
        print("Algorithm check failed")


# --------------------------------
# 5. SIMILARITY CHECKING
# --------------------------------

def similarity():

    print("\nSimilarity Checking")

    words1 = set(re.findall(r"[a-z_]+", codes))

    scores = []

    for code in real_code_dataset:

        words2 = set(re.findall(r"[a-z_]+", code))

        common = len(words1 & words2)
        total = len(words1 | words2)

        scores.append((common / total) * 100)

    print("Similarity Score:", round(max(scores), 2), "%")


# --------------------------------
# MAIN PROGRAM
# --------------------------------

generate()

testcase_testing()
checksyntax()
checkalgo()
similarity()
